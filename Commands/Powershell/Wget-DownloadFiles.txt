
# Download Files From PowerShell
********************************


# Particular Folder Download CMD.

-Uri: The URL of the file you want to download.

-OutFile: The full path where you want to save the file 

Invoke-WebRequest 
-Uri "https://download.oracle.com/graalvm/24/latest/graalvm-jdk-24_windows-x64_bin.zip" 
-OutFile "C:\SoftwareFiles\DownloadFromPowershell\graalvm-jdk-24_windows-x64_bin.zip"
